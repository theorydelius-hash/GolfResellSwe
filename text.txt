<script>
// Scroll till sektion
function scrollToSection(id){
  document.getElementById(id).scrollIntoView({behavior:'smooth'});
}

// Scroll till toppen
function scrollToTop(){ window.scrollTo({top:0, behavior:'smooth'}); }

// Golfbollar & Tillbehör data
const golfProducts = {
    "Titleist": [{name:"Pro V1", img:"https://cdn.pixabay.com/photo/2017/09/26/19/48/golf-ball-2782008_1280.jpg", desc:"Premium • Klass A • 12‑pack • Pris: 299kr"}],
    "TaylorMade": [{name:"TP5 Mix", img:"https://cdn.pixabay.com/photo/2016/11/18/15/35/golf-1837741_1280.jpg", desc:"Klass A • Blandade färger • Pris: 289kr"}],
    "Srixon": [{name:"Z‑Star Mix", img:"https://cdn.pixabay.com/photo/2017/08/28/16/42/golf-2692068_1280.jpg", desc:"Klass A/B • Blandade • Pris: 199kr"}],
    "Wilson": [{name:"Wilson Mix", img:"https://cdn.pixabay.com/photo/2016/02/23/18/23/golf-1213713_1280.jpg", desc:"Mix • 12‑pack • Pris: 199kr"}]
};

const accessories = [
    {name:"Golf Tee", img:"https://cdn.pixabay.com/photo/2016/04/30/11/18/golf-1362803_1280.jpg", desc:"Trä tees • 50-pack • Pris: 49kr"},
    {name:"Golf Handskar", img:"https://cdn.pixabay.com/photo/2015/07/01/11/46/golf-828040_1280.jpg", desc:"Läderhandskar • Pris: 129kr"}
];

// Skapa dropdown-knappar
const golfDropdown = document.getElementById('golfDropdown');
Object.keys(golfProducts).forEach(brand => {
    const a = document.createElement('a');
    a.textContent = brand;
    a.onclick = () => {
        showSection('golfSection');
        const container = document.getElementById('golfContainer');
        const infoBox = document.getElementById('ballInfo');
        container.innerHTML=''; infoBox.style.display='none';
        golfProducts[brand].forEach(ball => {
            const div = document.createElement('div');
            div.className='balls-card';
            div.innerHTML = `<img src="${ball.img}" alt="${ball.name}"><span>${ball.name}</span><button class="add-cart-btn" onclick="alert('${ball.name} tillagd')">Lägg i varukorg</button>`;
            div.onclick = () => { infoBox.innerHTML = `<strong>${ball.name}</strong><p>${ball.desc}</p>`; infoBox.style.display='block'; infoBox.scrollIntoView({behavior:'smooth'}); };
            container.appendChild(div);
        });
    };
    golfDropdown.appendChild(a);
});

const accDropdown = document.getElementById('accDropdown');
accessories.forEach(item => {
    const a = document.createElement('a');
    a.textContent = item.name;
    a.onclick = () => {
        showSection('accessorySection');
        const container = document.getElementById('accessoriesContainer');
        const infoBox = document.getElementById('accessoryInfo');
        container.innerHTML=''; infoBox.style.display='none';
        const div = document.createElement('div');
        div.className='accessory-card';
        div.innerHTML=`<img src="${item.img}" alt="${item.name}"><span>${item.name}</span><button class="add-cart-btn" onclick="alert('${item.name} tillagd')">Lägg i varukorg</button>`;
        div.onclick=()=>{infoBox.innerHTML=`<strong>${item.name}</strong><p>${item.desc}</p>`; infoBox.style.display='block'; infoBox.scrollIntoView({behavior:'smooth'});};
        container.appendChild(div);
    };
    accDropdown.appendChild(a);
});

// Dropdown toggle
document.getElementById('golfBtn').addEventListener('click',()=>{golfDropdown.classList.toggle('show'); accDropdown.classList.remove('show');});
document.getElementById('accBtn').addEventListener('click',()=>{accDropdown.classList.toggle('show'); golfDropdown.classList.remove('show');});

// Klick utanför stänger dropdown
window.addEventListener('click', e=>{
    if(!e.target.closest('#golfBtn')) golfDropdown.classList.remove('show');
    if(!e.target.closest('#accBtn')) accDropdown.classList.remove('show');
});
</script>
